WITH docs AS (
  SELECT 
    id AS record_id,
    director_id,
    document_id
  FROM acc_portal_directors_documents
  WHERE document_id = '63bf2181-f837-474e-a8cf-4cb6faa7aa6c'
),

json_source AS (
  SELECT
    record_id,
    director_id,
    document_id,
    (
      jsonb_build_object(
        '072b9710-45df-4a05-ba65-01b4ef29d61d',
        jsonb_build_object(
          'id', record_id::text,
          'document_id', document_id::text
        )
      )
    ) AS source_json
  FROM docs
),

stages_json AS (
  SELECT 
    jsonb_build_object(
      '1e6ebd85-db1e-4eb9-abb6-5fc145ac8caa', jsonb_build_object(
        'date', to_char(current_date, 'DD/MM/YYYY'),
        'status', 'completed',
        'completedAt', now()
      ),
      '9ccdbfa6-8d76-49db-b8d4-09742c734905', jsonb_build_object(
        'date', to_char(current_date, 'DD/MM/YYYY'),
        'status', 'completed',
        'completedAt', now()
      ),
      'a19b270e-6d1e-4e6d-b2bc-f21db72cde08', jsonb_build_object(
        'date', to_char(current_date, 'DD/MM/YYYY'),
        'status', 'completed',
        'completedAt', now()
      )
    ) AS stages
)

INSERT INTO tm_tasks (
  status,
  department_id,
  subdepartment_id,
  individual_id,
  document_id,
  source_document_id,
  stages,
  created_by
)
SELECT
  'in_progress' AS status,
  1 AS department_id,
  3 AS subdepartment_id,
  d.director_id AS individual_id,
  d.document_id,
  d.source_json,
  s.stages,
  'lynne' AS created_by
FROM json_source d
CROSS JOIN stages_json s;
