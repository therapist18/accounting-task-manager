/** Population - ALL ETA  **/
Showing 320 tasks

WITH docs AS (
  SELECT
    id,
    director_id,
    document_id,
    extracted_details,

    /* ---------------------------------------------
       APPLICATION NUMBER EXTRACTION (REPLACES PERMIT)
    ---------------------------------------------- */
    COALESCE(
      -- Direct keys
      extracted_details ->> 'application_number',
      extracted_details ->> 'Application_Number',
      extracted_details ->> 'Application Number',
      extracted_details ->> 'APPLICATION_NUMBER',
      extracted_details ->> 'APPLICATION NUMBER',

      -- Array fallback
      (
        SELECT elem ->> key
        FROM jsonb_each(extracted_details) AS t(section, value)
        CROSS JOIN LATERAL (
          SELECT elem
          FROM jsonb_array_elements(value) AS elem
        ) arr
        CROSS JOIN LATERAL (
          VALUES 
            ('application_number'),
            ('Application_Number'),
            ('Application Number'),
            ('APPLICATION_NUMBER'),
            ('APPLICATION NUMBER')
        ) AS keys(key)
        WHERE jsonb_typeof(value) = 'array'
          AND elem ->> key IS NOT NULL
        LIMIT 1
      )
    ) AS application_number

  FROM acc_portal_directors_documents
  WHERE document_id = 'f83594a0-7ddb-4fff-8d1b-ba9f1106486b'
),


/* -----------------------------------------------------
   GROUP BY director AND application number
------------------------------------------------------ */
grouped AS (
  SELECT 
    director_id,
    application_number,
    COUNT(*) AS count_per_director
  FROM docs
  GROUP BY director_id, application_number
)

SELECT 
  d.id,
  d.director_id,
  d.document_id,
  d.extracted_details,
  d.application_number,
  g.count_per_director,
  CASE WHEN g.count_per_director = 1 THEN 'unique'
       ELSE 'duplicate'
  END AS record_status
FROM docs d
JOIN grouped g 
    ON g.director_id = d.director_id
   AND g.application_number = d.application_number
ORDER BY 
  d.director_id,
  g.count_per_director DESC,
  d.application_number;
